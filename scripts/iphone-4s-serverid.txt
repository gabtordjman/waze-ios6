grep -v '^System.ServerId:' "$PREF" | grep -v '^GeoConfig.version:' > "$PREF.n"
echo "System.ServerId: 1" >> "$PREF.n"
echo "GeoConfig.version: 1" >> "$PREF.n"
mv "$PREF.n" "$PREF"

# Download URLs vers le PC
sed -i "s|http://75.101.158.200|http://$PC|g" "$PREF" 2>/dev/null
sed -i "s|http://75.101.158.200|http://$PC|g" "$APP/Waze.app/preferences" 2>/dev/null

grep -E 'ServerId|Download.Config' "$PREF"
rm -f "$APP"/Documents/session*
killall -9 Waze 2>/dev/null
echo "OK 4S — rouvre Waze"
