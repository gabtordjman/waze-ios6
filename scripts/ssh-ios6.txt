# SSH iOS 6 depuis le T480
#
# Raccourci (gère clé SSH + patch) :
#   sh scripts/patch-iphone.sh              # 4S @ 192.168.1.60
#   sh scripts/patch-iphone.sh 192.168.1.61
#
# Mot de passe = root du jailbreak (Cydia → OpenSSH, passwd root si besoin).
#
# Manuel :
ssh-keygen -f ~/.ssh/known_hosts -R 192.168.1.60
ssh -o HostKeyAlgorithms=+ssh-rsa -o PubkeyAcceptedAlgorithms=+ssh-rsa \
  -o StrictHostKeyChecking=accept-new root@192.168.1.60
